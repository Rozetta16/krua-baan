    โดนบล็อกบ่อย? ตั้ง relay ส่วนตัว (ฟรี ทีเดียวจบ)

    relay คือทางผ่านของเราเอง แอปจะลองช่องนี้ก่อนเสมอ ขั้นตอน: สมัคร Cloudflare (ฟรี) → Workers → สร้าง Worker → วางโค้ดด้านล่าง → Deploy → ก็อป address มาใส่ช่องนี้

    export default {
  async fetch(req) {
    const u = new URL(req.url).searchParams.get("url") || "";
    if (!/^https:\/\/(www\.)?wongnai\.com\//.test(u)) {
      return new Response("only wongnai.com allowed", { status: 400 });
    }
    const r = await fetch(u, { headers: { "User-Agent": "Mozilla/5.0 (compatible; krua-baan personal use)" } });
    const html = await r.text();
    return new Response(html, {
      headers: {
        "content-type": "text/html;charset=utf-8",
        "access-control-allow-origin": "*",
        "cache-control": "max-age=300"
      }
    });
  }
};
